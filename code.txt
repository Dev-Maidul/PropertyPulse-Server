// VerifyToken function
const verifyToken = (req, res, next) => {
  const token = req.cookies?.token;
  if (!token) {
    return res.status(401).send({ message: "unauthorized access" });
  }
  jwt.verify(token, process.env.ACCESS_TOKEN_SECRET, (err, decoded) => {
    if (err) {
      console.log(err);
      return res.status(401).send({ message: "unauthorized access" });
    }
    req.user = decoded;
    next();
  });
};

app.get("/properties", async (req, res) => {
  try {
    const { location = "", sort = "" } = req.query;
    const query = { status: "Verified" }; // Only verified properties

    // Search by location (case-insensitive)
    if (location) {
      query.location = { $regex: location, $options: "i" };
    }

    // Fetch all matching properties
    let properties = await propertiesCollection.find(query).toArray();

    // Sort by price
    if (sort === "asc" || sort === "desc") {
      properties = properties.sort((a, b) => {
        // Helper function to extract price from priceRange
        const getPrice = (priceRange) => {
          // Handle cases where priceRange is just a number (like "354")
          if (!isNaN(priceRange)) return Number(priceRange);
          
          // Handle price ranges (like "300-400")
          const prices = priceRange.split('-').map(Number);
          return prices[0]; // Use the first number for sorting
        };

        const aPrice = getPrice(a.priceRange);
        const bPrice = getPrice(b.priceRange);
        return sort === "asc" ? aPrice - bPrice : bPrice - aPrice;
      });
    }

    // Fetch agent images
    const agentEmails = properties.map((p) => p.agentEmail);
    const agents = await usersCollection
      .find({ email: { $in: agentEmails } })
      .toArray();

    // Map properties with agent images
    const propertiesWithAgentImage = properties.map((property) => {
      const agent = agents.find((a) => a.email === property.agentEmail);
      return {
        ...property,
        agentImage:
          agent?.image || "https://i.ibb.co/2d9dK0F/default-profile.png",
      };
    });

    res.status(200).json(propertiesWithAgentImage);
  } catch (err) {
    console.error("Error fetching properties:", err);
    res.status(500).json({ message: "Failed to fetch properties" });
  }
});

          _id: new ObjectId(offerId),
        });
        if (!offer) return res.status(404).send({ message: "Offer not found" });

        // Stripe expects amount in cents (for USD), for BDT use integer
        const amount = Number(offer.offerAmount);

        const paymentIntent = await stripe.paymentIntents.create({
          amount: amount,
          currency: "usd", // or 'bdt' if your Stripe supports it
          metadata: { offerId: offerId },
        });

        res.send({
          clientSecret: paymentIntent.client_secret,
        });
      } catch (err) {
        res.status(500).send({ message: "Failed to create payment intent" });
      }
    });
    //
    app.patch("/offers/payment/:offerId", async (req, res) => {
      try {
        const { transactionId } = req.body;
        const offerId = req.params.offerId;
        const result = await offersCollection.updateOne(
          { _id: new ObjectId(offerId) },
          { $set: { status: "bought", transactionId } }
        );
        res.send(result);
      } catch (err) {
        res
          .status(500)
          .send({ message: "Failed to update offer after payment" });
      }
    });
    // =========================agent related api==============
    // add properties to db
    app.post("/add-property", async (req, res) => {
      try {
        const result = await propertiesCollection.insertOne(req.body);
        res.send({ acknowledged: true, insertedId: result.insertedId });
      } catch (err) {
        res
          .status(500)
          .send({ acknowledged: false, message: "Failed to add property" });
      }
    });
    // GET /agent/properties?email=agent@example.com
    app.get("/agent/properties", async (req, res) => {
      const email = req.query.email;
      if (!email) return res.status(401).send({ message: "Email is required" });

      const properties = await propertiesCollection
        .find({ agentEmail: email })
        .toArray();

      const agent = await usersCollection.findOne({ email: email });

      const propertiesWithAgentImage = properties.map((property) => ({
        ...property,
        agentImage: agent?.image || null,
      }));

      res.send(propertiesWithAgentImage);
    });
    // get agent property by id
    app.get("/agent/property/:id", async (req, res) => {
      const id = req.params.id;
      const property = await propertiesCollection.findOne({
        _id: new ObjectId(id),
      });
      res.send(property);
    });
    // Get all sold properties for an agent
    app.get("/agent/sold-properties/:agentEmail", async (req, res) => {
      try {
        const agentEmail = req.params.agentEmail;
        // Find all offers for this agent where status is "bought"
        const soldOffers = await offersCollection
          .find({ agentEmail, status: "bought" })
          .sort({ createdAt: -1 })
          .toArray();
        res.send(soldOffers);
      } catch (err) {
        res.status(500).send({ message: "Failed to fetch sold properties" });
      }
    });

    // PUT /agent/property/:id
    app.put("/agent/property/:id", async (req, res) => {
      const id = req.params.id;
      const updateData = req.body;

      const result = await propertiesCollection.updateOne(
        { _id: new ObjectId(id) },
        { $set: updateData }
      );
      res.json(result);
    });
    // DELETE /agent/property/:id
    app.delete("/agent/property/:id", async (req, res) => {
      const id = req.params.id;
      const result = await propertiesCollection.deleteOne({
        _id: new ObjectId(id),
      });
      res.json(result);
    });
    // =================== User Related Api ===================
    app.post("/user", async (req, res) => {
      const userData = req.body;
      userData.role = "customer";
      userData.created_at = new Date().toISOString();
      userData.last_loggedIn = new Date().toISOString();

      userData.name = userData.name || "Anonymous";
      userData.image =
        userData.image || "https://i.ibb.co/2d9dK0F/default-profile.png";

      const query = { email: userData?.email };
      const alreadyExists = await usersCollection.findOne(query);

      if (!!alreadyExists) {
        const result = await usersCollection.updateOne(query, {
          $set: { last_loggedIn: new Date().toISOString() },
        });
        return res.send(result);
      }

      const result = await usersCollection.insertOne(userData);
      res.send(result);
    });
    // // get a user's role
    app.get("/user/role/:email", async (req, res) => {
      const email = req.params.email;
      const result = await usersCollection.findOne({ email });
      if (!result) return res.status(404).send({ message: "User Not Found." });
      res.send({ role: result?.role });
    });
    // user review
    app.get("/my-reviews/:userEmail", async (req, res) => {
      try {
        const userEmail = req.params.userEmail;
        const reviews = await reviewsCollection
          .find({ userEmail })
          .sort({ createdAt: -1 })
          .toArray();
        res.send(reviews);
      } catch (err) {
        res.status(500).send({ message: "Failed to fetch reviews" });
      }
    });
    // delete review
    app.delete("/my-reviews/:reviewId", async (req, res) => {
      try {
        const reviewId = req.params.reviewId;
        const result = await reviewsCollection.deleteOne({
          _id: new ObjectId(reviewId),
        });
        res.send(result);
      } catch (err) {
        res.status(500).send({ message: "Failed to delete review" });
      }
    });
    //===========================================
  } finally {
    // await client.close();
  }
}
run().catch(console.dir);
